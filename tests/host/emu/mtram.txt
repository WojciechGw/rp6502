# Run with --mtram --mtram-temp 45.5 --mtram-ds18b20 21.5 --mtram-ds18b20
# -10.125. Each task sees its own block at $8000, and the program halts with
# task 1 live, so the script's view of $8000 goes through task 1's map. The
# registers are in the window $FF00-$FF7F, and the old ones at $FEDD-$FEFF are
# plain RAM. The two sensors read 2150 and -1012 hundredths, the first
# device found is a DS18B20, and operation 4 is refused with XFER_ST_BADOP.

wait "T0=C3"
wait "T1=3C"
wait "TEMP=11C6"
wait "RST=01"
wait "SIG=6D7452"
wait "DS=020866FC0C"
wait "OW=002800"
wait "BAD=40"
wait "DONE"
peek $8000 $3C
peek $FF10 101
peek $FF7D $6D $74 $52
